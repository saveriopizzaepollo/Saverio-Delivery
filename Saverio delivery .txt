function bindLogin() {
  document.querySelectorAll(".login-role-btn").forEach(btn => {
    btn.addEventListener("click", () => {
      const u = STATE.utenti[btn.dataset.id];
      if (!u) return;
      loginSelezionato = u;
      loginFase = "pin";
      loginPin = "";
      loginErrore = false;
      render();
    });
  });
  document.getElementById("btn-riders")?.addEventListener("click", () => { loginFase = "listaRider"; render(); });
  document.getElementById("btn-back-ruoli")?.addEventListener("click", () => { loginFase = "ruoli"; render(); });
  document.getElementById("btn-back-pin")?.addEventListener("click", () => {
    loginFase = loginSelezionato?.ruolo === "rider" ? "listaRider" : "ruoli";
    loginPin = ""; loginErrore = false; render();
  });
  document.querySelectorAll(".pin-btn:not(.pin-invisible)").forEach(btn => {
    btn.addEventListener("click", () => {
      const d = btn.dataset.digit;
      if (d === "del") { loginPin = loginPin.slice(0,-1); loginErrore = false; render(); return; }
      loginPin += d;
      if (loginPin.length === 4) {
        if (loginSelezionato.pin === loginPin) {
          STATE.utenteLoggato = loginSelezionato;
          STATE.schermata = loginSelezionato.ruolo === "rider" ? "miei" : loginSelezionato.ruolo === "dispatch" ? "dispatch" : "dashboard";
          loginFase = "ruoli"; loginPin = ""; loginErrore = false;
          render();
        } else {
          loginErrore = true;
          setTimeout(() => { loginPin = ""; loginErrore = false; render(); }, 700);
          render();
        }
      } else { render(); }
    });
  });
}

}

document.getElementById(“f-telM”)?.addEventListener(“input”, e => f.telM = e.target.value);
document.getElementById(“f-indirizzoM”)?.addEventListener(“input”, e => f.indirizzoM = e.target.value);
document.getElementById(“f-note”)?.addEventListener(“input”, e => f.note = e.target.value);

document.getElementById(“sel-zona”)?.addEventListener(“change”, e => { f.zonaId = e.target.value; renderPartial(); });

document.querySelectorAll(”.prod-nome”).forEach(i => i.addEventListener(“input”, e => { const p = f.prodotti.find(x=>x.id===i.dataset.id); if(p) p.nome=e.target.value; }));

document.querySelectorAll(”.prod-prezzo”).forEach(i => i.addEventListener(“input”, e => { const p = f.prodotti.find(x=>x.id===i.dataset.id); if(p) { p.prezzo=e.target.value; renderPartial(); }}));

document.getElementById(“btn-conferma-ordine”)?.addEventListener(“click”, async () => {
const cliente = f.clienteId ? STATE.clienti[f.clienteId] : null;
if (!cliente && !f.nomeM) return toast(“Scegli o inserisci un cliente”, “err”);
if (f.prodotti.some(p => !p.nome)) return toast(“Inserisci nome ad ogni prodotto”, “err”);
const zona = Object.values(STATE.zone).find(z => z.id === f.zonaId);
const totProd = f.prodotti.reduce((s,p)=>s+(parseFloat(p.prezzo)||0)*(parseInt(p.qty)||1),0);

id: uid(), data: nowISO(), stato: “nuovo”,
clienteId: f.clienteId||””,
nomeCliente: cliente ? `${cliente.cognome} ${cliente.nome}`.trim() : f.nomeM,
telefono: cliente ? cliente.telefono||”” : f.telM,
indirizzo: cliente ? cliente.indirizzo||”” : f.indirizzoM,

zonaId: f.zonaId||””,

note: f.note, fonte: f.fonte,

function renderPartial() {
const page = document.querySelector(”.page”);
if (!page) return;
if (STATE.schermata === “nuovo”) { page.innerHTML = renderNuovoOrdine(); bindNuovoOrdine(); }
else if (STATE.schermata === “dispatch”) { page.innerHTML = renderDispatch(); bindDispatch(); }
}

function bindOrdini() {
if (STATE.schermata !== “ordini”) return;
document.querySelectorAll(”.filtro-btn”).forEach(b => b.addEventListener(“click”, () => { filtroOrdini = b.dataset.filtro; renderPartialOrdini(); }));
document.querySelectorAll(”.ordine-stato”).forEach(b => b.addEventListener(“click”, async () => {
await update(ref(db,`ordini/${b.dataset.id}`), {stato: b.dataset.stato});
toast(`Stato: ${STATI[b.dataset.stato].label}`);
}));
document.querySelectorAll(”.riassegna-btn”).forEach(b => b.addEventListener(“click”, () => {
const o = getOrdiniOggi().find(x => x.id === b.dataset.id);
STATE.modal = {tipo:“riassegna”, data:o};
render();
}));
}

function renderPartialOrdini() {
const page = document.querySelector(”.page”);
if (page && STATE.schermata === “ordini”) { page.innerHTML = renderOrdini(); bindOrdini(); }
}

function bindDispatch() {
if (STATE.schermata !== “dispatch”) return;
document.querySelectorAll(”.dispatch-card”).forEach(card => {
card.addEventListener(“click”, () => {
const id = card.dataset.id;
if (dispatchSelezionati[id]) delete dispatchSelezionati[id];
else dispatchSelezionati[id] = true;
renderPartial();
});
});
document.querySelectorAll(”.assegna-rider-btn”).forEach(b => b.addEventListener(“click”, async () => {
const rid = b.dataset.rid;
const ids = Object.keys(dispatchSelezionati);
if (ids.length === 0) return toast(“Seleziona almeno un ordine”, “err”);
for (const id of ids) {
await update(ref(db,`ordini/${id}`), {fattorinoId:rid, stato:“consegna”, oraUscita:nowISO(), riassegnaRichiesta:false});
}
dispatchSelezionati = {};
toast(`${ids.length} consegne assegnate ✓`);
renderPartial();
}));
document.querySelectorAll(”.riassegna-btn”).forEach(b => b.addEventListener(“click”, () => {
const o = getOrdiniOggi().find(x => x.id === b.dataset.id);
STATE.modal = {tipo:“riassegna”, data:o};
render();
}));
}

await update(ref(db,`ordini/${m.data.id}`), {stato:“pagato”, metodoPagamento:“Carta”, oraRientro:nowISO()});
toast(“Incasso registrato ✓”);
STATE.modal = null; render();
});

// Riassegna
document.querySelectorAll(”.riassegna-a-btn”).forEach(b => b.addEventListener(“click”, async () => {
await update(ref(db,`ordini/${m.data.id}`), {fattorinoId:b.dataset.rid, riassegnaRichiesta:false, oraUscita:nowISO()});
toast(“Riassegnato ✓”);
STATE.modal = null;
dispatchSelezionati = {};
render();
}));

// Salva cliente
document.getElementById(“modal-salva-cliente”)?.addEventListener(“click”, async () => {
const cognome = document.getElementById(“mc-cognome”)?.value?.trim();
if (!cognome) return toast(“Inserisci il cognome”, “err”);
const data = {
cognome, nome: document.getElementById(“mc-nome”)?.value||””,
telefono: document.getElementById(“mc-telefono”)?.value||””,
indirizzo: document.getElementById(“mc-indirizzo”)?.value||””,
};
if (m.tipo === “editCliente”) {
await update(ref(db,`clienti/${m.data.id}`), data);
toast(“Cliente aggiornato ✓”);
} else {
const id = uid();
await set(ref(db,`clienti/${id}`), {…data, id});
if (STATE.schermata === “nuovo” && formOrdine) {
formOrdine.clienteId = id;
window._ricercaCliente = “”;
}
toast(“Cliente salvato ✓”);
}
STATE.modal = null; render();
});

// Salva membro
document.getElementById(“modal-salva-membro”)?.addEventListener(“click”, async () => {
const nome = document.getElementById(“mm-nome”)?.value?.trim();
const pin = document.getElementById(“mm-pin”)?.value?.trim();
if (!nome) return toast(“Inserisci il nome”, “err”);
if (!pin || pin.length !== 4) return toast(“PIN deve essere di 4 cifre”, “err”);
const data = {
nome, cognome: document.getElementById(“mm-cognome”)?.value||””,
telefono: document.getElementById(“mm-telefono”)?.value||””,
ruolo: document.getElementById(“mm-ruolo”)?.value||“rider”,
pin, softpos: document.getElementById(“mm-softpos”)?.checked||false, attivo: true,
};
if (m.tipo === “editMembro”) {
await update(ref(db,`utenti/${m.data.id}`), data);
toast(“Aggiornato ✓”);
} else {
const id = uid();
await set(ref(db,`utenti/${id}`), {…data, id});
toast(“Aggiunto ✓”);
}
STATE.modal = null; render();
});
}

function getOrdiniOggi() {

function oggiKey() {
const d = new Date();
return `${d.getFullYear()}-${d.getMonth()+1}-${d.getDate()}`;
}
